    private fun requestSinglePermission(helper: PermissionHelper, permission: String, result: MethodChannel.Result) {
        when (val permissionResult = helper.requestPermission(permission)) {
            is PermissionResult.Granted -> result.success(true)
            is PermissionResult.Denied -> {
                requestCallbacks[permission.hashCode() and 0xFFFF] = result
            }
            is PermissionResult.Requesting -> {
                // Store the result callback to be called when the permission result is received
                Log.d("PermissionMaster", "Permission is being requested: $permission")
                requestCallbacks[permission.hashCode() and 0xFFFF] = result
            }
            is PermissionResult.ShowRationale -> {
                // Instead of showing rationale, we'll directly request the permission
                Log.d("PermissionMaster", "Should show rationale, but directly requesting permission: $permission")
                requestCallbacks[permission.hashCode() and 0xFFFF] = result
            }
            is PermissionResult.OpenSettings -> channel.invokeMethod("openAppSettings", null)
            is PermissionResult.Error -> {
                when (val error = permissionResult.error) {
                    is PermissionError.UnsupportedVersion -> {
                        // For version-specific errors, send a special message to Flutter
                        channel.invokeMethod("onUnsupportedVersion", mapOf(
                            "permission" to permission,
                            "message" to error.message,
                            "minSdkVersion" to error.minSdkVersion
                        ))
                        result.error("VERSION_ERROR", error.message, mapOf(
                            "minSdkVersion" to error.minSdkVersion,
                            "currentSdkVersion" to Build.VERSION.SDK_INT
                        ))
                    }
                    else -> result.error("PERMISSION_ERROR", error.message, null)
                }
            }
        }
    }

    private fun checkPermissionStatus(call: MethodCall, helper: PermissionHelper, result: MethodChannel.Result) {
        val permission = call.argument<String>("permission")
            ?: return result.error("INVALID_ARGUMENT", "Permission string cannot be null", null)

        // First check if we have a saved status in storage
        val savedStatusKey = "permission_${permission}_status"
        if (storage.contains(savedStatusKey)) {

    private fun requestPermissionsSequentially(helper: PermissionHelper, permissions: Array<String>, result: MethodChannel.Result) {

                return

            Log.d("PermissionMaster", "Processing permission at index $currentIndex: $permission")

            val requestCode = permission.hashCode() and 0xFFFF

            if (helper.isPermissionGranted(permission)) {

                results[permission] = "GRANTED"

                requestNextPermission()
                return

            requestCallbacks[requestCode] = object : MethodChannel.Result {

                        results[permission] = if (callbackResult) "GRANTED" else "DENIED"

                        // Handle map result (unlikely in this case)
                        val status = (callbackResult[permission] as? String) ?: "DENIED"
                        results[permission] = status

                        // Default to DENIED for unknown result types
                        results[permission] = "DENIED"
                    }

                    currentIndex++
                    requestNextPermission()

                    Log.e("PermissionMaster", "Error for permission $permission: $errorCode - $errorMessage")
                    results[permission] = "ERROR"
                    currentIndex++
                    requestNextPermission()

                override fun notImplemented() {

                    results[permission] = "NOT_IMPLEMENTED"
                    currentIndex++
                    requestNextPermission()

    private fun savePermissionStatus(permission: String, granted: Boolean) {

            val status = if (granted) "GRANTED" else "DENIED"
            Log.d("PermissionMaster", "Saving permission status for $permission: $status")

            // Save to GetStorage for Flutter access
            storage.write("permission_${permission}_status", status)

            // Save timestamp
            storage.write("permission_${permission}_time", System.currentTimeMillis())

            // For storage permissions, also save a general storage permission status
            if (permission.contains("STORAGE") || permission.contains("MEDIA")) {
                storage.write("permission_android.permission.READ_EXTERNAL_STORAGE_status", status)

            }

        }

    private fun handlePermissionResult(requestCode: Int, permissions: Array<out String>, grantResults: IntArray) {
        val result = requestCallbacks[requestCode] ?: return

        try {
            Log.d("PermissionMaster", "Handling permission result for requestCode: $requestCode, permissions: ${permissions.joinToString()}")

            // Check if this is a storage permission request on Android 13+
            val isStoragePermissionOnTiramisu = Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU &&
                permissions.any { it.startsWith("android.permission.READ_MEDIA_") }

            // Check if this is a single permission request from requestPermissionsSequentially
            val isSequentialRequest = permissions.size == 1

            if (isStoragePermissionOnTiramisu) {
                // For storage permissions on Android 13+
                val results = mutableMapOf<String, String>()
                var allGranted = true

                for (i in permissions.indices) {
                    val permission = permissions[i]
                    val grantResult = grantResults[i]
                    val granted = grantResult == PackageManager.PERMISSION_GRANTED
                    val status = if (granted) "GRANTED" else "DENIED"
                    results[permission] = status
                    Log.d("PermissionMaster", "Permission result for $permission: $status")

                    // Save the permission status to persistent storage
                    savePermissionStatus(permission, granted)

                    if (!granted) {
                        allGranted = false
                    }
                }

                // If this is a sequential request, we need to handle it differently
                if (isSequentialRequest) {
                    // Just return the result for this single permission
                    val granted = grantResults.isNotEmpty() && grantResults[0] == PackageManager.PERMISSION_GRANTED
                    result.success(granted)
                } else {
                    // For group requests, return the map of results
                    if (allGranted) {
                        // If all permissions are granted, return true for simplicity
                        result.success(true)
                    } else {
                        // Otherwise return the detailed map
                        result.success(results)
                    }
                }
            } else {
                // For other permissions or older Android versions
                val granted = grantResults.isNotEmpty() && grantResults[0] == PackageManager.PERMISSION_GRANTED

                // Save the permission status to persistent storage for all permissions
                for (i in permissions.indices) {
                    val permission = permissions[i]
                    val grantResult = if (i < grantResults.size) grantResults[i] else PackageManager.PERMISSION_DENIED
                    val isGranted = grantResult == PackageManager.PERMISSION_GRANTED
                    savePermissionStatus(permission, isGranted)
                }

                if (granted) {
                    result.success(true)
                } else {
                    activity?.let { activity ->
                        var shouldOpenSettings = false
                        for (permission in permissions) {
                            if (!ActivityCompat.shouldShowRequestPermissionRationale(activity, permission)) {
                                shouldOpenSettings = true
                                break
                            }
                        }
                        if (shouldOpenSettings) {
                            channel.invokeMethod("openAppSettings", null)
                        }
                        result.success(false)
                    } ?: result.success(false)
                }
            }
        } catch (e: Exception) {
            Log.e("PermissionMaster", "Error handling permission result: ${e.message}", e)
            try {
                result.error("PERMISSION_ERROR", "Error handling permission result: ${e.message}", null)
            } catch (e2: Exception) {
                Log.e("PermissionMaster", "Failed to send error result: ${e2.message}", e2)
            }
        } finally {
            requestCallbacks.remove(requestCode)
        }
    }
